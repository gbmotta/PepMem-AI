\begin{frame}{Descritores do peptídeo --- da sequência}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Carga líquida $q_p$ (pH\,$\sim$7)}
Soma dos resíduos ionizáveis:\\[0.1cm]
K, R $= +1$;\quad H $= +0{,}1$;\\
D, E $= -1$;\quad demais $= 0$.\\[0.15cm]
{\scriptsize Se houver carga anotada (\texttt{net\_charge}), ela prevalece.}
\end{block}
\vspace{0.2cm}
\begin{block}{Hidrofobicidade $h_p$}
Média Kyte--Doolittle por resíduo:
\[
h_p = \tfrac{1}{L}\sum_i H(\mathrm{AA}_i)
\]
{\scriptsize Ex.: I\,$=$\,4,5;\, L\,$=$\,3,8;\, K\,$=$\,$-$3,9.}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Momento hidrofóbico $\mu H_p$}
Eisenberg (hélice $\alpha$, ângulo $100^\circ$):
\[
\mu H = \tfrac{1}{L}\sqrt{
\big(\!\sum H_i\cos i\theta\big)^2
\!+\!
\big(\!\sum H_i\sin i\theta\big)^2}
\]
{\scriptsize Anfifilicidade: faces hidro\-fóbica / hidrofílica.}
\end{block}
\vspace{0.2cm}
\begin{block}{Importante}
Tudo é \textbf{in silico} a partir da sequência.\\
\textbf{Não} é medido na bancada.\\
{\scriptsize Código: \texttt{pepmem/peptide\_utils.py}, \texttt{pepmem/pmi.py}}
\end{block}
\end{column}
\end{columns}
\end{frame}

% ----------------------------------------------------------------
\begin{frame}{Membrana e PMI --- como se obtém}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{block}{Descritores da membrana}
Tabela de alvos (\texttt{membrane\_targets}):
\begin{itemize}
  \item $q_m$ (carga superficial)
  \item Fração aniônica, LPS, peptidoglicano
  \item Colesterol, ergosterol, envelope viral
\end{itemize}
{\scriptsize Tipologia OPM + regras do projeto\\
(Gram+/Gram$-$/fungo/célula normal\ldots).\\
$h_m$ na inferência: proxy fixo $0{,}5$.}
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{PMI (não é ensaio)}
\[
\begin{aligned}
\mathrm{PMI}=&\,\alpha\,q_p|q_m|+\beta\,h_p h_m\\
&+\gamma\,\mu H_p-\delta\,\mathrm{col}_m
\end{aligned}
\]
{\scriptsize $\alpha{=}1$; $\beta{=}0{,}5$; $\gamma{=}0{,}3$; $\delta{=}0{,}4$\\
(pesos fixos de projeto, não aprendidos).}
\end{block}
\vspace{0.15cm}
\begin{block}{Bancada vs PMI}
\textbf{Bancada / literatura:} MIC (rótulo).\\
\textbf{PMI:} índice calculado do par.\\
{\scriptsize PMI\_sel $=$ PMI$_{\mathrm{alvo}}-$PMI$_{\mathrm{normal}}$\\
(só no ranking, não no RF).}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Validação: LOO vs LOPO}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{LOO (por amostra)}
Tira \textbf{1 par} por vez.\\[0.15cm]
O mesmo peptídeo pode\\
ficar no treino (outro alvo).\\[0.15cm]
{\scriptsize AUC mais otimista\\
(vazamento por homologia).}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{LOPO (por peptídeo)}
Tira \textbf{todo} o peptídeo.\\[0.15cm]
Generalização para sequência\\
nunca vista.\\[0.15cm]
{\scriptsize Métrica principal +\\
probs OOF para calibração.}
\end{block}
\end{column}
\end{columns}
\vspace{0.35cm}
\begin{center}
{\small Família Stigmurin: análogos $\sim$94\% idênticos $\Rightarrow$ LOPO é essencial.}
\end{center}
\end{frame}

% ----------------------------------------------------------------
\begin{frame}{Calibração isotônica}
\begin{itemize}
  \item RF ordena bem, mas \texttt{predict\_proba} é \textbf{voto do ensemble}, não frequência real.
  \item Com $\sim$90 MICs, probs brutas tendem a polarizar (0/1).
  \item \textbf{IsotonicRegression} nas probs OOF do LOPO $\rightarrow$ $p_{\text{calibrada}}$.
  \item Preserva a ordem dos candidatos; alinha \% do dashboard ao LOPO.
\end{itemize}
\vspace{0.3cm}
\begin{block}{Inferência}
$p_{\text{bruta}} = \mathrm{RF.predict\_proba}$\quad$\rightarrow$\quad
$p_{\text{calibrada}} = \mathrm{calibrator.predict}([p_{\text{bruta}}])$
\end{block}
\end{frame}

  \item Poucos peptídeos distintos ($\sim$10 com MIC) --- LOPO ajuda, mas amostra é pequena
  \item Limiar 3,4\,µM é escolha de projeto, não padrão CLSI
  \item Hemólise existe nos pares, mas o RF classifica \textbf{atividade MIC}, não toxicidade direta
  \item Multimodal ainda é RF+ESM, não o encoder multimodal completo da proposta CNPq
  \item Predição \textbf{prioriza} ensaio; não substitui bancada
\end{itemize}
\end{frame}

\begin{frame}{Mensagens-chave}

  \item \textbf{Dados:} 90 MICs (bancada + literatura), rótulo MIC $\le$ 3,4\,µM
  \item \textbf{Descritores:} $q$, $h$, $\mu H$ da sequência; membrana tipificada; PMI \textbf{calculado} (não ensaiado)
  \item \textbf{Modelo:} Random Forest (Nível 1 InovAI) + PMI; ESM-2 no multimodal
  \item \textbf{Validação:} LOPO é a métrica honesta; isotônica calibra o dashboard
  \item \textbf{Resultado:} baseline LOPO AUC $\approx$ \textbf{0,88}; multimodal LOPO AUC $\approx$ \textbf{0,81}

\end{enumerate}
